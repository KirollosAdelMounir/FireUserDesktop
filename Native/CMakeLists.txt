cmake_minimum_required(VERSION 3.24)
project(FireUserLiveKit LANGUAGES CXX)

if(APPLE)
    enable_language(OBJCXX)
    add_library(fireuser_notifications SHARED platform_notifications.mm)
    target_compile_options(fireuser_notifications PRIVATE -fobjc-arc)
    target_link_libraries(fireuser_notifications PRIVATE "-framework Foundation" "-framework UserNotifications")
    install(TARGETS fireuser_notifications LIBRARY DESTINATION .)
    add_library(fireuser_screencapture SHARED screen_capture.mm)
    target_compile_features(fireuser_screencapture PRIVATE cxx_std_20)
    target_compile_options(fireuser_screencapture PRIVATE -fobjc-arc)
    target_link_libraries(fireuser_screencapture PRIVATE "-framework Foundation" "-framework ScreenCaptureKit" "-framework CoreMedia" "-framework CoreVideo")
    install(TARGETS fireuser_screencapture LIBRARY DESTINATION .)
endif()
find_package(LiveKit CONFIG REQUIRED)
add_library(fireuser_livekit SHARED fireuser_livekit.cpp desktop_input.cpp audio_devices.cpp)
target_include_directories(fireuser_livekit PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sdk/miniaudio")
target_compile_features(fireuser_livekit PRIVATE cxx_std_20)
target_link_libraries(fireuser_livekit PRIVATE LiveKit::livekit)
if(WIN32)
    target_link_libraries(fireuser_livekit PRIVATE user32 ole32 winmm)
elseif(APPLE)
    target_link_libraries(fireuser_livekit PRIVATE "-framework ApplicationServices" "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
else()
    find_package(X11 REQUIRED)
    if(NOT TARGET X11::Xtst)
        message(FATAL_ERROR "XTest development files are required (libxtst-dev).")
    endif()
    target_link_libraries(fireuser_livekit PRIVATE X11::X11 X11::Xtst ${CMAKE_DL_LIBS} pthread m)
endif()
set_target_properties(fireuser_livekit PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
if(APPLE)
    set_target_properties(fireuser_livekit PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
endif()
if(MSVC)
    target_compile_options(fireuser_livekit PRIVATE /W4 /EHsc /utf-8)
else()
    target_compile_options(fireuser_livekit PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS fireuser_livekit RUNTIME DESTINATION . LIBRARY DESTINATION .)
include(CTest)
if(BUILD_TESTING)
    add_executable(bridge_smoke bridge_smoke.cpp)
    target_link_libraries(bridge_smoke PRIVATE fireuser_livekit)
    if(FIREUSER_CURL_LIBRARY)
        target_link_libraries(bridge_smoke PRIVATE "${FIREUSER_CURL_LIBRARY}")
    endif()
    add_test(NAME bridge_abi_lifecycle COMMAND bridge_smoke)
endif()
