cmake_minimum_required(VERSION 3.24)
# Keep release builds on newer Macs compatible with the Sonoma target.
# This must precede project(), which initializes the compiler deployment flags.
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin" AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version")
endif()
project(FireUserLiveKit LANGUAGES CXX)

if(APPLE)
    enable_language(OBJCXX)
    add_library(fireuser_notifications SHARED platform_notifications.mm)
    target_compile_options(fireuser_notifications PRIVATE -fobjc-arc)
    target_link_libraries(fireuser_notifications PRIVATE "-framework Foundation" "-framework UserNotifications")
    install(TARGETS fireuser_notifications LIBRARY DESTINATION .)
    add_library(fireuser_screencapture SHARED screen_capture.mm)
    target_compile_features(fireuser_screencapture PRIVATE cxx_std_20)
    target_compile_options(fireuser_screencapture PRIVATE -fobjc-arc)
    target_link_libraries(fireuser_screencapture PRIVATE "-framework Foundation" "-framework ScreenCaptureKit" "-framework CoreMedia" "-framework CoreVideo" "-framework CoreGraphics")
    install(TARGETS fireuser_screencapture LIBRARY DESTINATION .)
endif()
# IDEs configure this directory directly, without the native build script's
# CMAKE_PREFIX_PATH. Prefer an explicitly configured SDK, then the pinned cache.
find_package(LiveKit CONFIG QUIET)
if(NOT LiveKit_FOUND)
    if(WIN32)
        set(_fireuser_os win)
    elseif(APPLE)
        set(_fireuser_os osx)
    else()
        set(_fireuser_os linux)
    endif()
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" _fireuser_processor)
    if(APPLE AND CMAKE_OSX_ARCHITECTURES)
        set(_fireuser_processor "${CMAKE_OSX_ARCHITECTURES}")
    elseif(CMAKE_GENERATOR_PLATFORM)
        string(TOLOWER "${CMAKE_GENERATOR_PLATFORM}" _fireuser_processor)
    endif()
    if(_fireuser_processor MATCHES "^(amd64|x86_64|x64)$")
        set(_fireuser_arch x64)
    elseif(_fireuser_processor MATCHES "^(aarch64|arm64)$")
        set(_fireuser_arch arm64)
    else()
        message(FATAL_ERROR "Unsupported native target ${_fireuser_processor}. Select a supported 64-bit compiler/architecture.")
    endif()
    set(_fireuser_rid "${_fireuser_os}-${_fireuser_arch}")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/native-runtimes.json" _fireuser_runtimes)
    string(JSON _fireuser_sdk_name ERROR_VARIABLE _fireuser_sdk_error GET "${_fireuser_runtimes}" "${_fireuser_rid}" name)
    if(NOT _fireuser_sdk_error AND NOT MINGW)
        set(_fireuser_sdk "${CMAKE_CURRENT_SOURCE_DIR}/sdk/${_fireuser_rid}/livekit-sdk-${_fireuser_sdk_name}-1.12.2")
        list(PREPEND CMAKE_PREFIX_PATH "${_fireuser_sdk}")
    endif()
    find_package(LiveKit CONFIG QUIET)
    if(NOT LiveKit_FOUND)
        message(FATAL_ERROR "LiveKit SDK not found for ${_fireuser_rid}. Run scripts/build-native.ps1 (Windows) or scripts/build-native.py (Linux/macOS) once to prepare the verified SDK cache, or set LiveKit_DIR to a compatible SDK's lib/cmake/LiveKit directory. MinGW requires a matching MinGW-built SDK.")
    endif()
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/sdk/miniaudio/miniaudio.h")
    message(FATAL_ERROR "The verified miniaudio header is missing. Run scripts/build-native.ps1 (Windows) or scripts/build-native.py (Linux/macOS) once to prepare native dependencies.")
endif()
add_library(fireuser_livekit SHARED fireuser_livekit.cpp desktop_input.cpp audio_devices.cpp screen_output.cpp)
target_include_directories(fireuser_livekit PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sdk/miniaudio")
target_compile_features(fireuser_livekit PRIVATE cxx_std_20)
target_link_libraries(fireuser_livekit PRIVATE LiveKit::livekit)
if(WIN32)
    target_link_libraries(fireuser_livekit PRIVATE user32 ole32 winmm dxgi)
elseif(APPLE)
    target_link_libraries(fireuser_livekit PRIVATE "-framework ApplicationServices" "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
else()
    find_package(X11 REQUIRED)
    if(NOT TARGET X11::Xtst)
        message(FATAL_ERROR "XTest development files are required (libxtst-dev).")
    endif()
    target_link_libraries(fireuser_livekit PRIVATE X11::X11 X11::Xtst ${CMAKE_DL_LIBS} pthread m)
endif()
set_target_properties(fireuser_livekit PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
if(APPLE)
    set_target_properties(fireuser_livekit PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
endif()
if(MSVC)
    target_compile_options(fireuser_livekit PRIVATE /W4 /EHsc /utf-8)
    get_target_property(_fireuser_sdk_configs LiveKit::livekit IMPORTED_CONFIGURATIONS)
    if("RELEASE" IN_LIST _fireuser_sdk_configs AND NOT "DEBUG" IN_LIST _fireuser_sdk_configs)
        # The pinned Windows SDK has a Release C++ ABI. Keep its CRT/STL layout
        # when debugging our bridge; Debug still retains symbols and /Od.
        set_property(TARGET fireuser_livekit PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
        target_compile_definitions(fireuser_livekit PRIVATE "$<$<CONFIG:Debug>:_ITERATOR_DEBUG_LEVEL=0>")
    endif()
else()
    target_compile_options(fireuser_livekit PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    # Direct IDE builds do not run the packaging script. Place the SDK DLLs
    # beside the bridge/test executable so native debugging needs no PATH edits.
    get_filename_component(_fireuser_sdk_root "${LiveKit_DIR}/../../.." ABSOLUTE)
    file(GLOB _fireuser_sdk_dlls "${_fireuser_sdk_root}/bin/*.dll")
    if(_fireuser_sdk_dlls)
        add_custom_command(TARGET fireuser_livekit POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different ${_fireuser_sdk_dlls} "$<TARGET_FILE_DIR:fireuser_livekit>"
            COMMAND_EXPAND_LISTS VERBATIM)
    endif()
endif()
install(TARGETS fireuser_livekit RUNTIME DESTINATION . LIBRARY DESTINATION .)
include(CTest)
if(BUILD_TESTING)
    add_executable(video_scaling_tests video_scaling_tests.cpp)
    target_compile_features(video_scaling_tests PRIVATE cxx_std_20)
    add_test(NAME video_scaling_pixels COMMAND video_scaling_tests)
    add_executable(video_source_tests video_source_tests.cpp)
    target_compile_features(video_source_tests PRIVATE cxx_std_20)
    target_link_libraries(video_source_tests PRIVATE LiveKit::livekit)
    if(FIREUSER_CURL_LIBRARY)
        target_link_libraries(video_source_tests PRIVATE "${FIREUSER_CURL_LIBRARY}")
    endif()
    if(MSVC)
        set_property(TARGET video_source_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
        target_compile_definitions(video_source_tests PRIVATE "$<$<CONFIG:Debug>:_ITERATOR_DEBUG_LEVEL=0>")
    endif()
    add_test(NAME video_source_i420 COMMAND video_source_tests)
    add_executable(bridge_smoke bridge_smoke.cpp)
    target_compile_features(bridge_smoke PRIVATE cxx_std_20)
    target_link_libraries(bridge_smoke PRIVATE fireuser_livekit)
    if(FIREUSER_CURL_LIBRARY)
        target_link_libraries(bridge_smoke PRIVATE "${FIREUSER_CURL_LIBRARY}")
    endif()
    add_test(NAME bridge_abi_lifecycle COMMAND bridge_smoke)
endif()
